public class Complexo 
{
    private double a, b;

    public /*void*/ Complexo (double a, double b)
    {
        this.a=a;
        this.b=b;
    }

    public Complexo mais (Complexo c) throws Exception
    {
        if (c==null) throw new Exception ("2º operando ausente");
        
        double aDoRet = this.a+c.a;
        double bDoRet = this.b+c.b;
        Complexo retorno = new Complexo (aDoRet,bDoRet);

        return retorno;
    }

    public Complexo menos (Complexo c) throws Exception
    {
        if (c==null) throw new Exception ("2º operando ausente");
        
        double aDoRet = this.a-c.a;
        double bDoRet = this.b-c.b;
        Complexo retorno = new Complexo (aDoRet,bDoRet);

        return retorno;
    }

    public Complexo vezes (Complexo c) throws Exception
    {
        if (c==null) throw new Exception ("2º operando ausente");
        
        double aDoRet = this.a*c.a - this.b*c.b;
        double bDoRet = this.a*c.b + c.a*this.b;
        Complexo retorno = new Complexo (aDoRet,bDoRet);

        return retorno;
    }

    public Complexo divididoPor (Complexo c) throws Exception
    {
        if (c==null) throw new Exception ("2º operando ausente");
        if (c.a==0 && c.b==0) throw new Exception ("Divisao por zero");
        
        double aDoRet = (this.a*c.a+this.b*c.b) / (Math.pow(c.a,2)+Math.sqr(c.b));
        double bDoRet = (c.a*this.b-this.a*c.b) / (c.a*c.a+Math.sqr(c.b));
        Complexo retorno = new Complexo (aDoRet,bDoRet);

        return retorno;
    }

    @Override
    public String toString ()
    {
        if (this.a==0 && this.b==0) return "0";

        String retorno="";
        if (this.a!=0) retorno+=this.a;
        if (this.a!=0 && this.b>0) retorno+='+';
        if (this.b==1)
            retorno+="i";
        else if (this.b!=0)
            retorno+=this.b+"i";
        return retorno;
    }

    @Override
    public boolean equals (Object obj)
    {
        if (obj==this) return true;
        if (obj==null) return false;
        if (obj.getClass()!=this.getClass()) return false;

        Complexo c = (Complexo)obj;
        if (this.a!=c.a) return false;
        if (this.b!=c.b) return false;

        return true;
    }

    @Override
    public int hashCode ()
    {
        int retorno=1;

        retorno = retorno*2 + ((Double)this.a).hashCode();
        retorno = retorno*2 + ((Double)this.b).hashCode();

        if (retorno<0) retorno=-retorno;
        return retorno;
    }
}
